ExtAPI.DataModel.Project.ProductVersion
import os

from Ansys.Mechanical.DataModel.Enums import GeometryImportPreference
from Ansys.ACT.Mechanical.Utilities import GeometryImportPreferences
#open template mechdb file
ExtAPI.DataModel.Project.Open(r"C:\Users\018219877\Downloads\LBPF\ANSYS\lpbfsim.mechdb")

import os

geopartdel=ExtAPI.DataModel.GeoData.Assemblies[0].AllParts

Model.DeleteParts(geopartdel)



lines = []

STEP_FILE = r"C:\Users\018219877\Downloads\LBPF\Solidworks\part_Rx0.0_Ry45.0_Rz0.0.step"

def log(msg):
    lines.append(str(msg))
    ExtAPI.Log.WriteMessage(str(msg))

model = ExtAPI.DataModel.Project.Model
step_file = STEP_FILE

if not os.path.exists(step_file):
    raise Exception("STEP file not found: " + step_file)

# ------------------------------------------------------------------
# 1) Import STEP geometry
# ------------------------------------------------------------------
geom_group = model.GeometryImportGroup
if geom_group is None:
    geom_group = model.AddGeometryImportGroup()

geom_import = geom_group.AddGeometryImport()

prefs = GeometryImportPreferences()

geom_import.Import(
    step_file,
    GeometryImportPreference.Format.Automatic,
    prefs
)

log("Imported STEP file: " + step_file)

# ------------------------------------------------------------------
# 2) Find imported bodies named "part" and "base"
# ------------------------------------------------------------------
all_bodies = ExtAPI.DataModel.GetObjectsByType(
    Ansys.Mechanical.DataModel.Enums.DataModelObjectCategory.Body
)

part_body = None
base_body = None

log("Bodies found after import:")
for body in all_bodies:
    log("  " + str(body.Name))

for body in all_bodies:
    name_l = body.Name.strip().lower()

    if part_body is None and name_l.endswith("|part"):
        part_body = body

    if base_body is None and name_l.endswith("|base"):
        base_body = body

if part_body is None:
    raise Exception('Could not find imported body ending with "|part".')

if base_body is None:
    raise Exception('Could not find imported body ending with "|base".')

log("Using part body: " + part_body.Name)
log("Using base body: " + base_body.Name)

sel_manager = ExtAPI.SelectionManager

part_sel = sel_manager.CreateSelectionInfo(
    Ansys.ACT.Interfaces.Common.SelectionTypeEnum.GeometryEntities
)
part_sel.Ids = [part_body.GetGeoBody().Id]

base_sel = sel_manager.CreateSelectionInfo(
    Ansys.ACT.Interfaces.Common.SelectionTypeEnum.GeometryEntities
)
base_sel.Ids = [base_body.GetGeoBody().Id]

am_process = DataModel.GetObjectsByName("AM Process")[0]
log("Using existing AMProcess")

for child in list(am_process.Children):
    if child.Name == "Body Fitted Cartesian":
        child.Delete()
        break

am_process.PartGeometry = part_sel
am_process.BuildGeometry = part_sel
am_process.BaseGeometry = base_sel

log("AM geometry scoped.")
